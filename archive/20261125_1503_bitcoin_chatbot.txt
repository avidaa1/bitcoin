function decryptBlob(blob,passphrase,onURL){

  const combined=new Uint8Array(data.length+tag.length);combined.set(data);combined.set(tag,data.length);
  const enc=new TextEncoder();
  return crypto.subtle.importKey('raw',enc.encode(passphrase),{name:'PBKDF2'},false,['deriveKey'])
    .then(k=>crypto.subtle.deriveKey({name:'PBKDF2',salt,iterations:100000,hash:'SHA-256'},k,{name:'AES-GCM',length:256},false,['decrypt']))

function lev(a,b){
  if(Math.abs(a.length-b.length)>2)return 9;
  const m=a.length,n=b.length,d=Array.from({length:m+1},(_,i)=>[i,...Array(n).fill(0)]);
  for(let j=0;j<=n;j++)d[0][j]=j;
  for(let i=1;i<=m;i++)for(let j=1;j<=n;j++)
    d[i][j]=Math.min(d[i-1][j]+1,d[i][j-1]+1,d[i-1][j-1]+(a[i-1]===b[j-1]?0:1));
  return d[m][n];
}
function detectTone(msg){

  if(/\b(hey|sup|yo|lol|wassup|gimme|u |ur )\b/.test(m))return'casual';
  return'neutral';
}

function smalltalkMatch(low,words){
  let best=null,bs=0;
  SMALLTALK.forEach(e=>{const s=scoreEntry(e,low,words);if(s>bs){bs=s;best=e;}});
  if(best&&bs>=6)return best.r[0];

function chatMatch(msg){
  const low=msg.toLowerCase().trim();
  chatSession.turns++; chatSession.asked.push(low);
  const tone=detectTone(msg); chatSession.lastTone=tone;
  const quest=questCheck(low);
  if(quest)return quest;
  if(quest===null && (t1Active||t2Active)) return {handled:true};
  if(isFrustrated(low))return {text:BACKOFF[Math.floor(Math.random()*BACKOFF.length)],link:null};
  const words=low.split(/\s+/).filter(w=>w.length>=2);
  const st=smalltalkMatch(low,words);
  if(st)return {text:st,link:null};
  const ranked=KB.filter(e=>!(e.r[0]&&e.r[0].startsWith('__QUEST')))
    .map(e=>({e,sc:scoreEntry(e,low,words)}))
    .filter(x=>x.sc>0)
    .sort((a,b)=>b.sc-a.sc);
  const media=mediaMatch(low);
  const bestScore=ranked.length?ranked[0].sc:0;
  if(media && bestScore<8){return {text:flavor(media.text,tone),link:media.link||null};}
  if(ranked.length && bestScore>=6)return compose(ranked,tone);
  return {text:REDIRECTS[Math.floor(Math.random()*REDIRECTS.length)],link:null};
}

function callClaude(prompt, maxTokens){
  var key = localStorage.getItem('avidaa_anthropic_key');
  if(!key){ return Promise.reject(new Error('NO_KEY')) }
  return fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': key,
      'anthropic-version': '2023-06-01',
      'anthropic-dangerous-direct-browser-access': 'true'
    },
    body: JSON.stringify({
      model: 'claude-sonnet-4-6',
      max_tokens: maxTokens || 700,
      messages: [{role:'user', content: prompt}]
    })
  })
  .then(function(r){ return r.json() })
  .then(function(data){
    if(data.error){ throw new Error(data.error.message || 'API error') }
    return (data.content || []).filter(function(b){return b.type==='text'}).map(function(b){return b.text}).join('\n');
  });
}

            + FIT_WEIGHTS.use[fitMatcherState.use];
  var fgNote = '';
  if(liveData.fearGreedVal !== undefined){
    if(liveData.fearGreedVal < 30){ score += 10; fgNote = ' Current Fear & Greed reading ('+liveData.fearGreedVal+', '+liveData.fearGreedLabel+') is in fear territory \u2014 historically a more constructive entry zone for long-horizon accumulation, adding +10 to your fit score.'; }
    else if(liveData.fearGreedVal > 70){ score -= 10; fgNote = ' Current Fear & Greed reading ('+liveData.fearGreedVal+', '+liveData.fearGreedLabel+') is in greed territory \u2014 elevated euphoria historically warrants extra caution, subtracting 10 from your fit score.'; }
    else { fgNote = ' Current Fear & Greed reading ('+liveData.fearGreedVal+', '+liveData.fearGreedLabel+') is neutral \u2014 no sentiment adjustment applied.'; }
  }
  score = Math.max(0, Math.min(100, score));
  var tier, tierColor;
  if(score >= 70){ tier = 'High Fit'; tierColor = 'var(--green)'; }
  else if(score >= 40){ tier = 'Moderate Fit'; tierColor = 'var(--gold)'; }
  else { tier = 'Low Fit'; tierColor = 'var(--red)'; }

  var resultEl = document.getElementById('fit-matcher-result');
  resultEl.style.display = 'block';
  resultEl.innerHTML =
    '<div style="display:flex;align-items:baseline;gap:12px;margin-bottom:12px">'
    + '<div style="font-family:var(--fontb);font-size:32px;font-weight:800;color:'+tierColor+'">'+score+'</div>'
    + '<div><div style="font-family:var(--fontb);font-weight:700;color:'+tierColor+'">'+tier+'</div><div style="font-size:10px;color:var(--text3)">out of 100 \u2014 computed instantly, offline</div></div>'
    + '</div>'
    + '<p style="color:var(--text2);font-size:12px;line-height:1.7">Based on '+FIT_COPY.horizon[fitMatcherState.horizon]+', '+FIT_COPY.risk[fitMatcherState.risk]+', used '+FIT_COPY.use[fitMatcherState.use]+'.'+fgNote+'</p>'
    + '<div style="font-size:10px;color:var(--text3);margin-top:10px">This is a deterministic rule-based score for educational reflection, not financial advice \u2014 recompute any time by clicking different answers.</div>';
}

function buildHodlersHistory(force){
  var el = document.getElementById('cred-history');
  if(!el) return;
  if(el.dataset.built && !force) return;
  el.dataset.built = '1';
  var history = [];
  try{ history = JSON.parse(localStorage.getItem(CRED_HISTORY_KEY)) || []; }catch(e){ history = []; }
  if(!history.length){
    el.innerHTML = '<p style="color:var(--text3);text-align:center;padding:16px;font-size:12px">No credibility tests run yet on this browser.</p>';
    return;
  }
  var html = '<div style="overflow-x:auto"><table class="timeline-table"><thead><tr>'

    + '</tr></thead><tbody>';
  history.forEach(function(h){
    var d = new Date(h.timestamp);
    var s = h.scores;

      + '<td style="white-space:nowrap;color:var(--text3);font-size:11px">'+d.toLocaleDateString()+' '+d.toLocaleTimeString()+'</td>'

      + '<td style="color:'+scoreColor(s.INTERPRETATION.score)+'">'+s.INTERPRETATION.score+'</td>'
      + '<td style="color:'+scoreColor(s.SECURITY.score)+'">'+s.SECURITY.score+'</td>'
      + '<td style="color:'+scoreColor(s.SENTIMENT.score)+'">'+s.SENTIMENT.score+'</td>'
      + '<td style="color:'+scoreColor(s.VALUE.score)+'">'+s.VALUE.score+'</td>'
      + '<td style="color:'+scoreColor(s.WORLD_TO_BTC.score)+'">'+s.WORLD_TO_BTC.score+'</td>'
      + '<td style="color:'+scoreColor(s.BTC_TO_WORLD.score)+'">'+s.BTC_TO_WORLD.score+'</td>'
      + '<td style="color:'+scoreColor(s.DAILY_USE.score)+'">'+s.DAILY_USE.score+'</td>'

  });
